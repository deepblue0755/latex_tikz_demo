\newcommand{\DrawWorkflow}{%
\begin{tikzpicture}[node distance=7mm and 14mm]

% ---------- Main Flow ----------
\node[humanbox] (human) {\textbf{\NodeHumanTop}\\[-0.4mm]\footnotesize \NodeHumanTopSub};
\node[humanbox, below=of human] (define) {\textbf{\NodeDefine}\\[-0.4mm]\footnotesize \NodeDefineSub};
\node[aibox, below=of define] (ai) {\textbf{\NodeAI}\\[-0.4mm]\footnotesize \NodeAISub};
\node[toolbox, below=of ai] (toolchain) {\textbf{\NodeToolchain}\\[-0.4mm]\footnotesize \NodeToolchainSub};
\node[toolbox, below=of toolchain] (tests) {\textbf{\NodeTests}\\[-0.4mm]\footnotesize \NodeTestsSub};
\node[toolbox, below=of tests] (runtime) {\textbf{\NodeRuntime}\\[-0.4mm]\footnotesize \NodeRuntimeSub};
\node[reviewbox, below=of runtime] (review) {\textbf{\NodeReview}\\[-0.4mm]\footnotesize \NodeReviewSub};
\node[decision, below=10mm of review] (decision) {\NodeDecision\\[-0.3mm]\footnotesize \NodeDecisionSub};

\node[failbox, below left=16mm and 28mm of decision] (failed) {\textbf{\NodeFailed}};
\node[passbox, below right=16mm and 28mm of decision] (passed) {\textbf{\NodePassed}};
\node[mergebox, below=of passed] (merge) {\textbf{\NodeMerge}\\[-0.4mm]\footnotesize \NodeMergeSub};

% ---------- Side Notes ----------
\node[infobox, right=22mm of toolchain.east, anchor=west] (note1) {
  {\bfseries\color{TitleBlue}\NoteCoreTitle}\\[1mm]
  \NoteCoreBody
};

\node[infobox, below=7mm of note1] (note2) {
  {\bfseries\color{TitleBlue}\NoteScopeTitle}\\[1mm]
  \NoteScopeBody
};

\node[infobox, below=7mm of note2] (note3) {
  {\bfseries\color{TitleBlue}\NoteToolsTitle}\\[1mm]
  \NoteToolsBody
};

% ---------- Color Legend: lower-right ----------
\node[legendbox, right=12mm of merge.east, anchor=west] (legend) {
  {\bfseries\color{TitleBlue}\LegendTitle}\\[1.2mm]
  \begin{tabular}{@{}ll@{}}
    \LegendSwatch{HumanBlue}{HumanBorder}{\LegendHuman} &
    \LegendSwatch{AIGreen}{AIBorder}{\LegendAI} \\[1.2mm]
    \LegendSwatch{ToolOrange}{ToolBorder}{\LegendTool} &
    \LegendSwatch{ReviewPurple}{ReviewBorder}{\LegendReview} \\[1.2mm]
    \LegendSwatch{PassGreen}{PassBorder}{\LegendPass} &
    \LegendSwatch{FailRed}{FailBorder}{\LegendFail} \\[1.2mm]
    \LegendSwatch{MergeGray}{MergeBorder}{\LegendMerge} & {}
  \end{tabular}
};

% ---------- Body Background ----------
\begin{scope}[on background layer]
\node[
  fit=(human)(define)(ai)(toolchain)(tests)(runtime)(review)(decision)(failed)(passed)(merge)(note1)(note2)(note3)(legend),
  rounded corners=4mm,
  draw=PanelBorder,
  fill=white,
  line width=0.8pt,
  inner xsep=8mm,
  inner ysep=8mm
] (bodybg) {};
\end{scope}

% ---------- Title: centered to the full body layout ----------
\node[titlebox, above=8mm of bodybg.north, anchor=south] (title) {
  {\Large\bfseries\color{TitleBlue}\ChartTitle}\\[1mm]
  {\normalsize\color{TitleBlue!85!black}\ChartSubtitle}
};

% ---------- Arrows ----------
\draw[flow] (human) -- (define);
\draw[flow] (define) -- (ai);
\draw[flow] (ai) -- (toolchain);
\draw[flow] (toolchain) -- (tests);
\draw[flow] (tests) -- (runtime);
\draw[flow] (runtime) -- (review);
\draw[flow] (review) -- (decision);

% Decision branches: exactly two orthogonal segments,
% first horizontal, then vertical into Failed / Passed.
\coordinate (failturn) at (failed.north |- decision.west);
\coordinate (passturn) at (passed.north |- decision.east);
\draw[flow] (decision.west) -- (failturn) -- node[pos=0.55,left=1mm,labeltag]{\BranchNo} (failed.north);
\draw[flow] (decision.east) -- (passturn) -- node[pos=0.55,right=1mm,labeltag]{\BranchYes} (passed.north);
\draw[flow] (passed) -- (merge);

% Feedback loop: Failed -> AI
\coordinate[left=28mm of ai] (fbleft);
\draw[feedback] (failed.west) -- ++(-10mm,0) |- (fbleft) |- (ai.west);
\node[labeltag, above right=1mm and 1mm of fbleft, text=FailBorder] {\FeedbackText};

\end{tikzpicture}%
}
